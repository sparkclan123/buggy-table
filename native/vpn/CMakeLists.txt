cmake_minimum_required(VERSION 3.20)
project(buggy_vpn LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3 REQUIRED)
find_package(Threads REQUIRED)
if(NOT DEFINED VPN_DEPS)
  message(FATAL_ERROR "Build with scripts/build-vpn.py")
endif()
file(GLOB LWIP_CORE CONFIGURE_DEPENDS "${VPN_DEPS}/lwip/src/core/*.c" "${VPN_DEPS}/lwip/src/core/ipv4/*.c" "${VPN_DEPS}/lwip/src/core/ipv6/*.c")
add_library(vpn_lwip STATIC ${LWIP_CORE})
target_include_directories(vpn_lwip PUBLIC . "${VPN_DEPS}/lwip/src/include")
add_library(vpn_lz4 STATIC "${VPN_DEPS}/lz4/lib/lz4.c")
target_include_directories(vpn_lz4 PUBLIC "${VPN_DEPS}/lz4/lib")
add_library(vpn_json STATIC "${VPN_DEPS}/jsoncpp/src/lib_json/json_reader.cpp" "${VPN_DEPS}/jsoncpp/src/lib_json/json_value.cpp" "${VPN_DEPS}/jsoncpp/src/lib_json/json_writer.cpp")
target_include_directories(vpn_json PUBLIC "${VPN_DEPS}/jsoncpp/include")
add_executable(buggy-vpn main.cpp stack_test.c "${VPN_DEPS}/openvpn3/openvpn/crypto/data_epoch.cpp")
target_include_directories(buggy-vpn BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/overrides" "${VPN_DEPS}/../overrides")
target_include_directories(buggy-vpn PRIVATE "${VPN_DEPS}/openvpn3" "${VPN_DEPS}/asio/asio/include" "${VPN_DEPS}/xxhash")
target_compile_definitions(buggy-vpn PRIVATE ASIO_STANDALONE USE_ASIO USE_OPENSSL HAVE_LZ4 HAVE_JSONCPP OPENVPN_EXTERNAL_TUN_FACTORY XXH_INLINE_ALL)
target_link_libraries(buggy-vpn PRIVATE vpn_lwip vpn_lz4 vpn_json OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
if(WIN32)
  target_compile_definitions(buggy-vpn PRIVATE _WIN32_WINNT=0x0A00 ASIO_DISABLE_LOCAL_SOCKETS _CRT_SECURE_NO_WARNINGS)
  target_compile_options(buggy-vpn PRIVATE /bigobj)
  target_link_libraries(buggy-vpn PRIVATE ws2_32 crypt32 iphlpapi fwpuclnt wininet setupapi rpcrt4 wtsapi32 bcrypt)
  target_compile_options(buggy-vpn PRIVATE /guard:cf)
  target_link_options(buggy-vpn PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
elseif(APPLE)
  target_link_libraries(buggy-vpn PRIVATE "-framework CoreFoundation" "-framework IOKit" "-framework SystemConfiguration" "-framework CoreServices")
endif()
if(NOT MSVC)
  foreach(target buggy-vpn vpn_lwip vpn_lz4 vpn_json)
    target_compile_options(${target} PRIVATE -fstack-protector-strong)
  endforeach()
endif()
option(VPN_SANITIZE "Address and undefined behavior sanitizers" OFF)
if(VPN_SANITIZE AND NOT MSVC)
  foreach(target buggy-vpn vpn_lwip vpn_lz4 vpn_json)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
  target_link_options(buggy-vpn PRIVATE -fsanitize=address,undefined)
endif()
